
pipeline {
    agent any

    triggers {
        pollSCM('H/2 * * * *')
    }

    stages {

        stage('Install Dependencies') {
            steps {
                echo 'Installation des dependances Python'
                bat 'python -m pip install -r requirements.txt'
            }
        }

        stage('Run Tests') {
            steps {
                echo 'Execution des tests Python'
                bat 'python -m pytest tests/ -v'
            }
        }
    }
}
